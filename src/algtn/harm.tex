
\subsubsection{Números harmônicos}

$$ H_n = 1 + \dfrac{1}{2} + \dots + \dfrac{1}{n}$$
$$ \lim\limits_{n \to \infty} H_n - \ln(n) = \gamma \approx 0.5772$$

\prob{1}{harm-1} Dado $n$ inteiro positivo, qual um upper bound pro tamanho do conjunto
$$D_n = \left\{ \left\lfloor \dfrac{n}{k} \right\rfloor \, \vert \, 1 \leq k \leq n \right\}$$

\textbf{Solução:} $2 \sqrt{n}$. Isso porque se $i > \sqrt{n}$,
$\left\lfloor \dfrac{n}{i} \right\rfloor < \dfrac{n}{i} < \sqrt{n}$.

\prob{2}{harm-2} Seja $q \in S_n$. Seja $l$ o menor inteiro positivo tal que $\left\lfloor n/l \right\rfloor = q$. Seja
$r$ o maior inteiro positivo tal que $\left\lfloor n/r \right\rfloor = q$. Então,
$$ r = \left\lfloor \dfrac{n}{\left\lfloor \dfrac{n}{l} \right\rfloor} \right\rfloor$$ 

\textbf{Solução:} A prova segue da aplicação do algoritmo da divisão em $l$ e num $l'$ que satisfaz 
a propriedade que $l$ e $r$ satisfazem (i.e. $l \leq l' \leq r$). E isso acaba provando (sse) que $l' \leq n/q$.
Então o $r$ seria $\left\lfloor n/q \right\rfloor$ e substituir $q$ por $\left\lfloor n/l \right\rfloor$ nos dá o que queríamos.

Baseado nesses fatos, o seguinte for itera pelo conjunto $D_n$.
